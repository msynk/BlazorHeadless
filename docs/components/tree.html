Tree View

A hierarchical list — file explorers, navigation, nested settings — with expand / collapse, single, multiple and checkbox selection, lazy loading and typeahead.

File explorer

The WAI-ARIA Tree View pattern. Click a row to select it (@@bind-SelectedValue) or the chevron (BhTreeItemToggle) to expand without selecting. Keyboard: Up / Down move, Right expands or enters a folder, Left collapses or returns to the parent, Home / End jump, * expands every sibling, Enter (or double click) fires OnItemActivate, and typing a name jumps to it (typeahead, matched against Label). aria-level, aria-setsize, aria-posinset and the roving tabindex are computed for you; disabled items are skipped.

Data-driven (recursive)

Render the tree from a model with a recursive RenderFragment<Node> helper. Each BhTreeItem works out its depth from where it is rendered, so the markup is the same at every level. Supplying @@bind-ExpandedValues makes expansion controlled, so plain buttons can expand or collapse everything.

Example

Razor

<div class="tree-actions">
    <button type="button" class="tree-button" @onclick="ExpandAllDocs">Expand all</button>
    <button type="button" class="tree-button" @onclick="CollapseAllDocs">Collapse all</button>
</div>

<BhTree TValue="string" aria-label="Documentation" class="tree"
        @bind-SelectedValue="selectedDoc"
        @bind-ExpandedValues="expandedDocs">
    @foreach (var node in docs)
    {
        @DocItem(node)
    }
</BhTree>

<p class="tree-output">
    Selected: <strong>@(selectedDoc ?? "(none)")</strong> ·
    Expanded: <strong>@expandedDocs.Count</strong>
</p>

@code {
    private sealed record Node(string Id, string Name, Node[] Children);

    private readonly Node[] docs =
    [
        new("start", "Getting started",
        [
            new("installation", "Installation", []),
            new("quick-start", "Quick start", []),
        ]),
        new("components", "Components",
        [
            new("overlays", "Overlays",
            [
                new("dialog", "Dialog", []),
                new("popover", "Popover", []),
                new("tooltip", "Tooltip", []),
            ]),
            new("collections", "Collections",
            [
                new("listbox", "Listbox", []),
                new("tree", "Tree View", []),
                new("grid", "Grid", []),
            ]),
        ]),
        new("styling", "Styling",
        [
            new("plain-css", "Plain CSS", []),
            new("tailwind", "Tailwind", []),
        ]),
    ];

    private string? selectedDoc = "tree";
    private IReadOnlyCollection<string> expandedDocs = ["components", "collections"];

    private void ExpandAllDocs() => expandedDocs = Branches(docs).ToArray();

    private void CollapseAllDocs() => expandedDocs = [];

    private static IEnumerable<string> Branches(IEnumerable<Node> nodes) =>
        nodes.Where(n => n.Children.Length > 0).SelectMany(n => Branches(n.Children).Prepend(n.Id));

    // One template renders every level — it simply calls itself for the children.
    private RenderFragment<Node> DocItem => node =>
        @<BhTreeItem TValue="string" Value="@node.Id" Label="@node.Name" class="tree-item">
            <div class="tree-row">
                <BhTreeItemToggle TValue="string" class="tree-toggle">▸</BhTreeItemToggle>
                <span>@node.Name</span>
                @if (node.Children.Length > 0)
                {
                    <span class="tree-count">@node.Children.Length</span>
                }
            </div>
            @if (node.Children.Length > 0)
            {
                <BhTreeGroup TValue="string" class="tree-group">
                    @foreach (var child in node.Children)
                    {

                    <BhTreeItemToggle TValue="string" class="inline-flex size-5 flex-none items-center justify-center rounded text-xs text-zinc-400 transition-transform duration-150 hover:bg-zinc-200 hover:text-zinc-900 data-state-open:rotate-90 data-[leaf]:invisible">▸</BhTreeItemToggle>
                    <span data-checked="@CheckedAttr(state.Checked)" aria-hidden="true"
                          class="inline-flex size-4 flex-none items-center justify-center rounded border-[1.5px] border-zinc-400 bg-white text-[0.7rem] font-bold leading-none text-white data-[checked=true]:border-blue-600 data-[checked=true]:bg-blue-600 data-[checked=mixed]:border-blue-600 data-[checked=mixed]:bg-blue-600">@(state.Checked switch { true => "✓", null => "–", _ => "" })</span>
                    <span>@node.Name</span>
                </div>
                @if (node.Children.Length > 0)
                {
                    <BhTreeGroup TValue="string">
                        @foreach (var child in node.Children)
                        {
                            @FeatureItem(child)
                        }
                    </BhTreeGroup>
                }
            </Template>
        </BhTreeItem>;
}

Lazy loading

Mark folders whose children are not loaded yet with HasChildren="true" and supply LoadChildren. It runs the first time a folder expands; meanwhile the item reports aria-busy="true" (and IsLoading in its Template) so you can show a spinner. The folder opens once the children are rendered.

Example

Razor

<BhTree TValue="string" aria-label="Cloud storage" class="tree"
        LoadChildren="LoadChildrenAsync">
    @foreach (var bucket in buckets)
    {
        @RemoteItem(bucket)
    }
</BhTree>

@code {
    private sealed record RemoteEntry(string Path, string Name, bool IsFolder);

    private readonly RemoteEntry[] buckets =
    [
        new("media", "media-bucket", true),
        new("logs", "logs-bucket", true),
        new("backups", "backups-bucket", true),
    ];

    private readonly Dictionary<string, RemoteEntry[]> loadedChildren = new();

    // Called once per folder, the first time it expands.
    private async Task LoadChildrenAsync(string path)
    {
        await Task.Delay(800); // simulate a network round-trip

        if (path.Contains('/'))
        {
            loadedChildren[path] =
            [
                new($"{path}/january.csv", "january.csv", false),
                new($"{path}/february.csv", "february.csv", false),
            ];
        }
        else
        {
            loadedChildren[path] =
            [
                new($"{path}/2025", "2025", true),
                new($"{path}/2026", "2026", true),
                new($"{path}/manifest.json", "manifest.json", false),
            ];
        }

        StateHasChanged();
    }

    private RenderFragment<RemoteEntry> RemoteItem => entry =>
        @<BhTreeItem TValue="string" Value="@entry.Path" Label="@entry.Name" HasChildren="entry.IsFolder" class="tree-item">
            <Template Context="state">
                <div class="tree-row">
                    <BhTreeItemToggle TValue="string" class="tree-toggle">▸</BhTreeItemToggle>
                    <span class="tree-icon">@(entry.IsFolder ? (state.IsExpanded ? "📂" : "📁") : "📄")</span>
                    <span>@entry.Name</span>
                    @if (state.IsLoading)
                    {
                        <span class="tree-meta">Loading…</span>
                        <span class="tree-spinner" aria-hidden="true"></span>
                    }
                </div>
                @if (loadedChildren.TryGetValue(entry.Path, out var children))
                {
                    <BhTreeGroup TValue="string" class="tree-group">
                        @foreach (var child in children)
                        {
                            @RemoteItem(child)
                        }
                    </BhTreeGroup>
                }
            </Template>
        </BhTreeItem>;
}
