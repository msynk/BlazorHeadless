Combobox

An accessible typeable autocomplete with consumer-driven filtering.

Basic Combobox

Type to filter, use Up / Down to navigate, Enter to select, Escape to close. The combobox emits an OnQueryChange event; you do the filtering in your own code and pass only the matching options as children.

Example

Razor

<BhCombobox TValue="string"
           Value="@selectedFruit"
           OnValueChange="v => selectedFruit = v"
           OnQueryChange="HandleFruitQuery"
           DisplayValue="v => v ?? string.Empty"
           class="combobox">
    <div class="combobox-input-wrapper">
        <BhComboboxInput TValue="string"
                        Placeholder="Pick a fruit…"
                        class="combobox-input" />
        <BhComboboxButton TValue="string" class="combobox-button" Context="b">
            <span class="combobox-chevron @(b.IsOpen ? "open" : "")">▾</span>
        </BhComboboxButton>
    </div>
    <BhComboboxOptions TValue="string" class="combobox-options">
        @if (filteredFruits.Count == 0)
        {
            <li class="combobox-empty">No matches.</li>
        }
        else
        {
            @foreach (var fruit in filteredFruits)
            {
                <BhComboboxOption TValue="string" Value="@fruit" class="combobox-option">
                    @fruit
                </BhComboboxOption>
            }
        }
    </BhComboboxOptions>
</BhCombobox>

@code {
    private string? selectedFruit;
    private string[] allFruits = new[] {
        "Apple", "Apricot", "Banana", "Blueberry", "Cherry",
        "Coconut", "Cranberry", "Grape", "Kiwi", "Lemon",
        "Lime", "Mango", "Orange", "Papaya", "Peach",
        "Pear", "Pineapple", "Plum", "Raspberry", "Strawberry"
    };
    private List<string> filteredFruits = new();

    private void HandleFruitQuery(string q) =>
        filteredFruits = string.IsNullOrWhiteSpace(q)
            ? allFruits.ToList()
            : allFruits.Where(f => f.Contains(q, StringComparison.OrdinalIgnoreCase)).ToList();

    protected override void OnInitialized() => filteredFruits = allFruits.ToList();
}

Multi-select with chips

Set Multiple="true" to allow many selections. Selected values are shown as chips above the input (each with a remove button). The panel stays open after each pick.

Example

Razor

<BhCombobox TValue="string" Multiple="true"
           Values="@pickedTags"
           OnValuesChange="v => pickedTags = v"
           OnQueryChange="HandleTagQuery"
           class="combobox">
    <div class="combobox-input-wrapper combobox-input-wrapper-multi">
        @foreach (var tag in pickedTags)
        {
            <span class="combobox-chip">
                @tag
                <button type="button"
                        class="combobox-chip-remove"
                        @onclick="() => RemoveTag(tag)"
                        aria-label="@($"Remove {tag}")">×</button>
            </span>
        }
        <BhComboboxInput TValue="string"
                        Placeholder="@(pickedTags.Count == 0 ? "Pick tags…" : "")"
                        class="combobox-input combobox-input-inline" />
        <BhComboboxButton TValue="string" class="combobox-button" Context="b">
            <span class="combobox-chevron @(b.IsOpen ? "open" : "")">▾</span>
        </BhComboboxButton>
    </div>
    <BhComboboxOptions TValue="string" class="combobox-options">
        @if (filteredTags.Count == 0)
        {
            <li class="combobox-empty">No matches.</li>
        }
        else
        {
            @foreach (var tag in filteredTags)
            {
                <BhComboboxOption TValue="string" Value="@tag" class="combobox-option" Context="o">
                    <span class="combobox-tick">@(o.IsSelected ? "✓" : "")</span>
                    <span>@tag</span>
                </BhComboboxOption>
            }
        }
    </BhComboboxOptions>
</BhCombobox>

@code {
    private IReadOnlyCollection<string> pickedTags = Array.Empty<string>();
    private string[] allTags = new[] {
        "design", "engineering", "marketing", "sales", "support",
        "research", "product", "operations", "finance", "legal"
    };
    private List<string> filteredTags = new();

    private void HandleTagQuery(string q) =>
        filteredTags = string.IsNullOrWhiteSpace(q)
            ? allTags.ToList()
            : allTags.Where(t => t.Contains(q, StringComparison.OrdinalIgnoreCase)).ToList();

    private void RemoveTag(string tag) =>
        pickedTags = pickedTags.Where(t => t != tag).ToList();

    protected override void OnInitialized() => filteredTags = allTags.ToList();
}

Object values with rich content

Bind to objects rather than strings. Provide a DisplayValue function that converts the selected object back to the input text, and render rich content inside each option.

Example

Razor

public sealed record Person(string Name, string Email)
{
    public string Initials =>
        string.Concat(Name.Split(' ', StringSplitOptions.RemoveEmptyEntries)
                          .Take(2)
                          .Select(p => p[0]));
}

<BhCombobox TValue="Person"
           Value="@selectedPerson"
           OnValueChange="v => selectedPerson = v"
           OnQueryChange="HandlePersonQuery"
           DisplayValue="p => p?.Name ?? string.Empty"
           class="combobox">
    <div class="combobox-input-wrapper">
        <BhComboboxInput TValue="Person"
                        Placeholder="Search people…"
                        class="combobox-input" />
        <BhComboboxButton TValue="Person" class="combobox-button" Context="b">
            <span class="combobox-chevron @(b.IsOpen ? "open" : "")">▾</span>
        </BhComboboxButton>
    </div>
    <BhComboboxOptions TValue="Person" class="combobox-options">
        @if (filteredPeople.Count == 0)
        {
            <li class="combobox-empty">No matches.</li>
        }
        else
        {
            @foreach (var person in filteredPeople)
            {
                <BhComboboxOption TValue="Person" Value="@person" class="combobox-option" Context="o">
                    <div class="person-row">
                        <span class="person-avatar">@person.Initials</span>
                        <div class="person-info">
                            <strong>@person.Name</strong>
                            <span>@person.Email</span>
                        </div>
                        @if (o.IsSelected) { <span class="combobox-tick">✓</span> }
                    </div>
                </BhComboboxOption>
            }
        }
    </BhComboboxOptions>
</BhCombobox>

@code {
    private Person? selectedPerson;
    private Person[] allPeople = new[]
    {
        new Person("Alice Anderson",  "alice@example.com"),
        new Person("Benjamin Brown",  "ben@example.com"),
        new Person("Caterina Costa",  "cat@example.com"),
        new Person("Daniel Davis",    "dan@example.com"),
        new Person("Eleanor Evans",   "ellie@example.com"),
        new Person("Felix Foster",    "felix@example.com"),
        new Person("Genevieve Gomez", "gen@example.com"),
        new Person("Henry Hughes",    "henry@example.com"),
    };
    private List<Person> filteredPeople = new();

    private void HandlePersonQuery(string q) =>
        filteredPeople = string.IsNullOrWhiteSpace(q)
            ? allPeople.ToList()
            : allPeople.Where(p =>
                p.Name.Contains(q, StringComparison.OrdinalIgnoreCase) ||
                p.Email.Contains(q, StringComparison.OrdinalIgnoreCase)).ToList();

    protected override void OnInitialized() => filteredPeople = allPeople.ToList();
}

    private readonly Person[] allPeople =
    {
        new("Alice Anderson",  "alice@example.com"),
        new("Benjamin Brown",  "ben@example.com"),
        new("Caterina Costa",  "cat@example.com"),
        new("Daniel Davis",    "dan@example.com"),
        new("Eleanor Evans",   "ellie@example.com"),
        new("Felix Foster",    "felix@example.com"),
        new("Genevieve Gomez", "gen@example.com"),
        new("Henry Hughes",    "henry@example.com"),
    };
    private List<Person> filteredPeople = new();

    private void HandlePersonQuery(string q) =>
        filteredPeople = string.IsNullOrWhiteSpace(q) ? allPeople.ToList()
            : allPeople.Where(p =>
                p.Name.Contains(q, StringComparison.OrdinalIgnoreCase) ||
                p.Email.Contains(q, StringComparison.OrdinalIgnoreCase)).ToList();

    protected override void OnInitialized() => filteredPeople = allPeople.ToList();
}

Anchor Positioning

Set Anchor on BhComboboxOptions to automatically position the dropdown relative to the input using position: fixed. The panel flips to the opposite side when there's not enough space, shifts to stay within the viewport, and auto-updates on scroll/resize. No position: relative on the parent is needed.

Example

Razor

@* Anchor positions the dropdown relative to the input automatically.
   No position:relative on parent or position:absolute on panel needed.
   The panel uses position:fixed and auto-updates on scroll/resize. *@

<BhCombobox TValue="string"
           Value="@selectedFruit"
           OnValueChange="v => selectedFruit = v"
           OnQueryChange="HandleFruitQuery"
           DisplayValue="v => v ?? string.Empty"
           class="combobox-anchored">
    <div class="combobox-input-wrapper">
        <BhComboboxInput TValue="string" Placeholder="Pick a fruit…" class="combobox-input" />
        <BhComboboxButton TValue="string" class="combobox-button">▾</BhComboboxButton>
    </div>
    <BhComboboxOptions TValue="string"
                      Anchor="@(new BhAnchorOptions { To = "bottom start", Gap = 4 })"
                      class="combobox-options-anchored">
        @foreach (var fruit in filteredFruits)
        {
            <BhComboboxOption TValue="string" Value="@fruit" class="combobox-option">
                @fruit
            </BhComboboxOption>
        }
    </BhComboboxOptions>
</BhCombobox>

@* Available placements: top, top start, top end,
   bottom, bottom start, bottom end,
   left, left start, left end,
   right, right start, right end

   The panel auto-flips to the opposite side if there's not enough space. *@
