Listbox

An accessible custom select with single/multi-select, full keyboard navigation, and typeahead.

Basic Listbox

Click the button or press Space / Enter / arrow keys to open. Use Up / Down to navigate, Home / End to jump, Enter / Space to select, Escape to close, or simply type to jump to an option. ARIA wiring (combobox, listbox, aria-activedescendant) is handled for you.

Example

Razor

<BhListbox TValue="string" Value="@favPerson"
          OnValueChange="v => favPerson = v"
          class="listbox">
    <BhListboxButton TValue="string" class="listbox-button" Context="b">
        <span>@(string.IsNullOrEmpty(b.Value) ? "Select a person…" : b.Value)</span>
        <span class="listbox-chevron @(b.IsOpen ? "open" : "")">▾</span>
    </BhListboxButton>
    <BhListboxOptions TValue="string" class="listbox-options">
        @foreach (var name in people)
        {
            <BhListboxOption TValue="string" Value="@name" class="listbox-option">
                @name
            </BhListboxOption>
        }
    </BhListboxOptions>
</BhListbox>

@code {
    private string? favPerson;

    private string[] people = new[] {
        "Alice", "Benjamin", "Caterina", "Daniel",
        "Eleanor", "Felix", "Genevieve", "Henry"
    };
}

Multi-select

Set Multiple="true" to allow selecting many values. The panel stays open after each selection. The button label here renders the joined names.

Example

Razor

<BhListbox TValue="string" Multiple="true"
          Values="@favColors"
          OnValuesChange="v => favColors = v"
          class="listbox">
    <BhListboxButton TValue="string" class="listbox-button" Context="b">
        <span>
            @(b.Values.Count == 0 ? "Pick colours…" : string.Join(", ", b.Values))
        </span>
        <span class="listbox-chevron @(b.IsOpen ? "open" : "")">▾</span>
    </BhListboxButton>
    <BhListboxOptions TValue="string" class="listbox-options">
        @foreach (var c in colors)
        {
            <BhListboxOption TValue="string" Value="@c" class="listbox-option" Context="o">
                <span class="check-mark">@(o.IsSelected ? "✓" : "")</span>
                <span>@c</span>
            </BhListboxOption>
        }
    </BhListboxOptions>
</BhListbox>

@code {
    private IReadOnlyCollection<string> favColors = Array.Empty<string>();
    private string[] colors = new[] { "Red", "Green", "Blue", "Yellow", "Purple", "Orange" };
}

Disabled Options

Mark individual options with Disabled="true". They render with data-disabled for CSS, are skipped by arrow-key nav, and ignore typeahead matches.

Example

Razor

<BhListbox TValue="string" DefaultValue="standard" class="listbox">
    <BhListboxButton TValue="string" class="listbox-button" Context="b">
        <span>@(b.Value ?? "Pick a plan")</span>
        <span class="listbox-chevron @(b.IsOpen ? "open" : "")">▾</span>
    </BhListboxButton>
    <BhListboxOptions TValue="string" class="listbox-options">
        <BhListboxOption TValue="string" Value="standard"   class="listbox-option">Standard</BhListboxOption>
        <BhListboxOption TValue="string" Value="pro"        class="listbox-option">Pro</BhListboxOption>
        <BhListboxOption TValue="string" Value="enterprise" Disabled="true" class="listbox-option">
            Enterprise (coming soon)
        </BhListboxOption>
    </BhListboxOptions>
</BhListbox>

/* Hooks emitted when an option is disabled */

Custom Option Content

Options can render any markup. Use the data-active and data-selected hooks to drive the highlighted and selected styles. Provide a Label string when your content is rich so typeahead still matches sensibly.

Example

Razor

<BhListbox TValue="string" Value="@country"
          OnValueChange="v => country = v"
          class="listbox">
    <BhListboxButton TValue="string" class="listbox-button" Context="b">
        <span>@(LookupCountry(b.Value)?.Name ?? "Pick a country")</span>
        <span class="listbox-chevron @(b.IsOpen ? "open" : "")">▾</span>
    </BhListboxButton>
    <BhListboxOptions TValue="string" class="listbox-options">
        @foreach (var c in countries)
        {
            <BhListboxOption TValue="string" Value="@c.Code"
                            Label="@c.Name"
                            class="listbox-option">
                <span class="country-flag">@c.Flag</span>
                <span class="country-name">@c.Name</span>
                <span class="country-code">@c.Code</span>
            </BhListboxOption>
        }
    </BhListboxOptions>
</BhListbox>

@code {
    private record Country(string Code, string Name, string Flag);

    private string? country = "fr";

    private Country[] countries = new[]
    {
        new Country("us", "United States", "🇺🇸"),
        new Country("gb", "United Kingdom", "🇬🇧"),
        new Country("fr", "France",         "🇫🇷"),
        new Country("de", "Germany",        "🇩🇪"),
        new Country("jp", "Japan",          "🇯🇵"),
        new Country("br", "Brazil",         "🇧🇷"),
    };

    private Country? LookupCountry(string? code) =>
        code is null ? null : countries.FirstOrDefault(c => c.Code == code);
}

CSS

/* Rich option layout: flag + name + code */

.country-code {
    font-size: 0.7rem;
    color: #71717a;
    text-transform: uppercase;
    letter-spacing: 0.05em;
}

/* Provide Label="@c.Name" on each option so typeahead matches the visible
   country name rather than the country code stored as the value. */

Anchor Positioning

Set Anchor on BhListboxOptions to automatically position the dropdown relative to the button using position: fixed. The panel flips to the opposite side when there's not enough space, shifts to stay within the viewport, and auto-updates on scroll/resize. The CSS variable --button-width is exposed so you can match the dropdown width to the trigger. No position: relative on the parent is needed.

Example

Razor

@* Anchor positions the dropdown relative to the button automatically.
   No position:relative on parent or position:absolute on panel needed.
   The panel uses position:fixed and auto-updates on scroll/resize. *@

<BhListbox TValue="string" Value="@person"
          OnValueChange="v => person = v"
          class="listbox-anchored">
    <BhListboxButton TValue="string" class="listbox-button" Context="b">
        <span>@(b.Value ?? "Select a person…")</span>
        <span class="listbox-chevron @(b.IsOpen ? "open" : "")">▾</span>
    </BhListboxButton>
    <BhListboxOptions TValue="string"
                     Anchor="@(new BhAnchorOptions { To = "bottom start", Gap = 4 })"
                     class="listbox-options-anchored">
        @foreach (var name in people)
        {
            <BhListboxOption TValue="string" Value="@name" class="listbox-option">
                @name
            </BhListboxOption>
        }
    </BhListboxOptions>
</BhListbox>

@* Available placements: top, top start, top end,
   bottom, bottom start, bottom end,
   left, left start, left end,
   right, right start, right end

   The panel auto-flips to the opposite side if there's not enough space.
   Use --button-width CSS variable to match the trigger width. *@

CSS

/* When using Anchor, the panel is position:fixed by the JS engine.
   No position:relative on parent needed. */
.listbox-anchored { display: inline-block; min-width: 14rem; }

.listbox-options-anchored {
    z-index: 9999;
    margin: 0;
    padding: 0.25rem;
    list-style: none;
    background: #fff;
    border: 1px solid #e4e4e7;
    border-radius: 0.5rem;
    box-shadow: 0 6px 16px rgba(0, 0, 0, 0.08);
    max-height: 14rem;
    overflow-y: auto;
    /* Match the button width using the exposed CSS variable */
    width: var(--button-width);
}

/* Explicit hidden override — required when the panel has display set */
.listbox-options-anchored[hidden] { display: none; }

Form Submission

Set Name on the listbox and the selection participates in plain HTML form posts. In multi-select mode a separate hidden input is rendered for each value.

Example

Razor

<form @onsubmit="HandleFormSubmit"
      @onsubmit:preventDefault="true"
      style="display:flex; flex-direction:column; gap:0.75rem; align-items:flex-start;">
    <BhListbox TValue="string" Name="role" DefaultValue="user" class="listbox">
        <BhListboxButton TValue="string" class="listbox-button" Context="b">
            <span>@(b.Value)</span>
            <span class="listbox-chevron @(b.IsOpen ? "open" : "")">▾</span>
        </BhListboxButton>
        <BhListboxOptions TValue="string" class="listbox-options">
            <BhListboxOption TValue="string" Value="user"  class="listbox-option">User</BhListboxOption>
            <BhListboxOption TValue="string" Value="admin" class="listbox-option">Admin</BhListboxOption>
            <BhListboxOption TValue="string" Value="owner" class="listbox-option">Owner</BhListboxOption>
        </BhListboxOptions>
    </BhListbox>

    <button type="submit" class="btn btn-primary">Submit</button>

    @if (lastSubmitted is not null)
    {
        <span class="demo-output">@lastSubmitted</span>
    }
</form>

@code {
    private string? lastSubmitted;

    private void HandleFormSubmit()
    {
        lastSubmitted = "(submitted — when single-select, posts role=<value>; when multi-select, posts a separate field per value)";
    }
}

@* When Multiple="true" a separate hidden input is rendered per selected value. *@
