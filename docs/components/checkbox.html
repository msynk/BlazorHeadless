Checkbox

An accessible tri-state checkbox primitive with full keyboard support and zero visual opinion.

Basic Checkbox (Uncontrolled)

Renders a native <button> with role="checkbox" and aria-checked. Click or press Space to toggle. DefaultChecked seeds the initial state.

Example

Razor

<BhCheckbox class="checkbox" Context="c">
    @if (c.IsChecked) { <span class="checkbox-tick">✓</span> }
</BhCheckbox>

<BhCheckbox DefaultChecked="true" class="checkbox" Context="c">
    @if (c.IsChecked) { <span class="checkbox-tick">✓</span> }
</BhCheckbox>

CSS

/* All styles are consumer-authored — the library ships zero CSS */
.checkbox {
    /* Reset native button */
    all: unset;
    box-sizing: border-box;

    /* Box */
    display: inline-flex;
    align-items: center;
    justify-content: center;
    width: 1.15rem;
    height: 1.15rem;
    border: 1px solid #d4d4d8;
    border-radius: 0.25rem;
    background: #fff;
    color: #fff;
    cursor: pointer;
    transition: background 120ms ease, border-color 120ms ease;
    flex-shrink: 0;
}

.checkbox:hover                 { border-color: #a1a1af; }
.checkbox:focus-visible         { outline: 2px solid #2563eb; outline-offset: 2px; }

/* Hooks emitted by BhCheckbox */
.checkbox[data-state="checked"],
.checkbox[data-state="indeterminate"] {
    background: #2563eb;
    border-color: #2563eb;
}

.checkbox-tick,
.checkbox-dash {
    font-size: 0.85rem;
    line-height: 1;
}

Controlled

Drive state externally with Checked + OnCheckedChange.

Example

Razor

<BhCheckbox Checked="@acceptedTerms"
           OnCheckedChange="v => acceptedTerms = v"
           class="checkbox" Context="c">
    @if (c.IsChecked) { <span class="checkbox-tick">✓</span> }
</BhCheckbox>

@code {
    private bool acceptedTerms;
}

CSS

/* Reuses the .checkbox styles from the Basic demo.
   Controlled mode adds no special CSS hooks — data-state simply reflects
   whatever Checked value you bind to. */

Razor + Tailwind

<div class="flex items-center gap-4">
    <BhCheckbox Checked="@acceptedTerms" OnCheckedChange="v => acceptedTerms = v"
                class="inline-flex h-[1.15rem] w-[1.15rem] flex-shrink-0 items-center justify-center rounded border border-zinc-300 bg-white text-white cursor-pointer transition-colors outline-none hover:border-zinc-400 focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-blue-600 data-state-checked:bg-blue-600 data-state-checked:border-blue-600 data-disabled:opacity-50 data-disabled:cursor-not-allowed"
                Context="c">
        @if (c.IsChecked) { <span class="text-sm leading-none">✓</span> }
    </BhCheckbox>
    <span class="text-sm text-zinc-500">
        Terms accepted: <strong class="text-zinc-900">@(acceptedTerms ? "yes" : "no")</strong>
    </span>
    <BhButton OnClick="() => acceptedTerms = !acceptedTerms"
              class="inline-flex items-center gap-2 rounded-lg border border-zinc-200 bg-white px-3 py-1.5 text-sm font-medium text-zinc-900 cursor-pointer hover:bg-zinc-100">
        Toggle from outside
    </BhButton>
</div>

@code {
    private bool acceptedTerms;
}

Indeterminate (Select All pattern)

Setting Indeterminate="true" emits aria-checked="mixed" and data-state="indeterminate". The classic use-case is a "select all" checkbox that reflects partial child selection. Clicking an indeterminate checkbox advances it to checked.

Example

Razor

<div style="display:flex; flex-direction:column; gap:0.5rem;">
    <label class="checkbox-label">
        <BhCheckbox Checked="@SelectAllChecked"
                   Indeterminate="@SelectAllIndeterminate"
                   OnCheckedChange="HandleSelectAll"
                   class="checkbox" Context="c">
            @if (c.IsIndeterminate)  { <span class="checkbox-dash">–</span> }
            else if (c.IsChecked)    { <span class="checkbox-tick">✓</span> }
        </BhCheckbox>
        <strong>Select all</strong>
    </label>

    <div style="display:flex; flex-direction:column; gap:0.4rem; margin-left:1.6rem;">
        @for (var i = 0; i < items.Length; i++)
        {
            var captured = i;
            <label class="checkbox-label">
                <BhCheckbox Checked="@items[captured]"
                           OnCheckedChange="v => HandleItemChange(captured, v)"
                           class="checkbox" Context="c">
                    @if (c.IsChecked) { <span class="checkbox-tick">✓</span> }
                </BhCheckbox>
                <span>@($"Item {captured + 1}")</span>
            </label>
        }
    </div>
</div>

@code {
    private bool[] items = new[] { false, false, false };

    private bool SelectAllChecked       => items.All(x => x);
    private bool SelectAllIndeterminate => items.Any(x => x) && !SelectAllChecked;

    private void HandleSelectAll(bool checkAll)
    {
        for (var i = 0; i < items.Length; i++) items[i] = checkAll;
    }

    private void HandleItemChange(int index, bool value) => items[index] = value;
}

CSS

/* Hooks emitted when Indeterminate="true" */
.checkbox[data-state="indeterminate"] {
    background: #2563eb;
    border-color: #2563eb;
}

/* The dash glyph rendered inside the box for the mixed state */
.checkbox-dash {
    font-size: 1rem;
    font-weight: 700;
    line-height: 0;
}

Razor + Tailwind

<div class="flex flex-col gap-2">
    <label class="inline-flex items-center gap-2 cursor-pointer text-sm select-none">
        <BhCheckbox Checked="@SelectAllChecked" Indeterminate="@SelectAllIndeterminate" OnCheckedChange="HandleSelectAll"
                    class="inline-flex h-[1.15rem] w-[1.15rem] flex-shrink-0 items-center justify-center rounded border border-zinc-300 bg-white text-white cursor-pointer transition-colors outline-none hover:border-zinc-400 focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-blue-600 data-state-checked:bg-blue-600 data-state-checked:border-blue-600 data-state-indeterminate:bg-blue-600 data-state-indeterminate:border-blue-600 data-disabled:opacity-50 data-disabled:cursor-not-allowed"
                    Context="c">
            @if (c.IsIndeterminate) { <span class="text-base leading-none font-bold">–</span> }
            else if (c.IsChecked) { <span class="text-sm leading-none">✓</span> }
        </BhCheckbox>
        <strong>Select all</strong>
    </label>
    <div class="flex flex-col gap-1.5 ml-6">
        @for (var i = 0; i < items.Length; i++)
        {
            var captured = i;
            <label class="inline-flex items-center gap-2 cursor-pointer text-sm select-none">
                <BhCheckbox Checked="@items[captured]" OnCheckedChange="v => HandleItemChange(captured, v)"
                            class="inline-flex h-[1.15rem] w-[1.15rem] flex-shrink-0 items-center justify-center rounded border border-zinc-300 bg-white text-white cursor-pointer transition-colors outline-none hover:border-zinc-400 focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-blue-600 data-state-checked:bg-blue-600 data-state-checked:border-blue-600 data-disabled:opacity-50 data-disabled:cursor-not-allowed"
                            Context="c">
                    @if (c.IsChecked) { <span class="text-sm leading-none">✓</span> }
                </BhCheckbox>
                <span>Item @(captured + 1)</span>
            </label>
        }
    </div>
</div>

@code {
    private bool[] items = new[] { false, false, false };

    private bool SelectAllChecked => items.All(x => x);
    private bool SelectAllIndeterminate => items.Any(x => x) && !SelectAllChecked;

    private void HandleSelectAll(bool checkAll)
    {
        for (var i = 0; i < items.Length; i++) items[i] = checkAll;
    }

    private void HandleItemChange(int index, bool value) => items[index] = value;
}

Form Submission

Set Name to render an adjacent <input type="hidden"> so the checkbox state participates in plain form posts. The hidden input is only present when the checkbox is checked or indeterminate, mirroring native checkbox behaviour.

Example

Razor

<form method="post">
    <label class="checkbox-label">
        <BhCheckbox Name="agree" Value="yes"
                   class="checkbox" Context="c">
            @if (c.IsChecked) { <span class="checkbox-tick">✓</span> }
        </BhCheckbox>
        <span>I agree to the terms</span>
    </label>

    <button type="submit">Submit</button>
</form>

@* When checked, posts:  agree=yes *@

CSS

/* Reuses the .checkbox / .checkbox-label styles from earlier demos.
   No additional CSS is needed — when Name is set BhCheckbox renders an
   <input type="hidden"> sibling that is invisible by default. */

Razor + Tailwind

<form @onsubmit="HandleFormSubmit" @onsubmit:preventDefault="true"
      class="flex flex-col items-start gap-2">
    <label class="inline-flex items-center gap-2 cursor-pointer text-sm select-none">
        <BhCheckbox Name="agree" Value="yes"
                    class="inline-flex h-[1.15rem] w-[1.15rem] flex-shrink-0 items-center justify-center rounded border border-zinc-300 bg-white text-white cursor-pointer transition-colors outline-none hover:border-zinc-400 focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-blue-600 data-state-checked:bg-blue-600 data-state-checked:border-blue-600 data-disabled:opacity-50 data-disabled:cursor-not-allowed"
                    Context="c">
            @if (c.IsChecked) { <span class="text-sm leading-none">✓</span> }
        </BhCheckbox>
        <span>I agree to the terms</span>
    </label>
    <button type="submit"
            class="inline-flex items-center gap-2 rounded-lg border border-blue-600 bg-blue-600 px-3 py-1.5 text-sm font-medium text-white cursor-pointer hover:bg-blue-700">
        Submit
    </button>
    @if (lastSubmitted is not null)
    {
        <span class="text-sm text-zinc-500">@lastSubmitted</span>
    }
</form>

@code {
    private string? lastSubmitted;

    private void HandleFormSubmit() =>
        lastSubmitted = "(submitted — when checked, the form posts agree=yes)";
}
