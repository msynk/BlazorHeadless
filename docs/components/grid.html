Grid / TreeGrid

An interactive two-dimensional widget — data tables, spreadsheets-lite, file trees with columns, calendars — with roving cell focus, row selection, sorting and expandable rows.

Sortable data grid

The WAI-ARIA Data Grid pattern on a real <table> (As="table" / "thead" / "tbody" / "tr" / "th" / "td"). The grid is a single tab stop: arrow keys move between cells, Home / End jump to the ends of the row, Ctrl + Home / End to the corners, PageUp / PageDown move by PageSize rows. Column headers are cells too — click one or press Enter on it and its OnActivate toggles the sort, which Sort announces as aria-sort.

Example

Razor

<div class="datagrid-frame">
    <BhGrid As="table" aria-label="Planets" class="datagrid">

                    Type<span class="datagrid-sort" aria-hidden="true">▲</span>
                </BhGridCell>
                <BhGridCell As="th" Header="true" class="datagrid-sortable num"

@code {
    private sealed record Planet(string Name, string Type, int DiameterKm, int Moons);

    private readonly Planet[] planets =
    [
        new("Mercury", "Terrestrial", 4_879, 0),
        new("Venus", "Terrestrial", 12_104, 0),
        new("Earth", "Terrestrial", 12_756, 1),
        new("Mars", "Terrestrial", 6_792, 2),
        new("Jupiter", "Gas giant", 142_984, 95),
        new("Saturn", "Gas giant", 120_536, 146),
        new("Uranus", "Ice giant", 51_118, 28),
        new("Neptune", "Ice giant", 49_528, 16),
    ];

    private string sortColumn = "name";
    private BhSortDirection sortDirection = BhSortDirection.Ascending;

    // The sorted column announces aria-sort; the others pass None.
    private BhSortDirection SortOf(string column) => sortColumn == column ? sortDirection : BhSortDirection.None;

    private void SortBy(string column)
    {
        sortDirection = sortColumn == column && sortDirection == BhSortDirection.Ascending
            ? BhSortDirection.Descending
            : BhSortDirection.Ascending;
        sortColumn = column;
    }

    private IEnumerable<Planet> SortedPlanets
    {
        get
        {
            Func<Planet, object> key = sortColumn switch
            {
                "type" => p => p.Type,
                "diameter" => p => p.DiameterKm,
                "moons" => p => p.Moons,
                _ => p => p.Name,
            };
            return sortDirection == BhSortDirection.Descending ? planets.OrderByDescending(key) : planets.OrderBy(key);
        }
    }
}

Multi-row selection

SelectionMode="BhGridSelectionMode.Multiple" with @@bind-SelectedKeys (every data row needs a Key). Click toggles a row and Shift + click selects a range; Space toggles the focused row, Shift + Space or Shift + Up / Down extend the selection, Ctrl + A selects every row. Rows get aria-selected, so the check boxes are pure CSS; the header box uses OnActivate to select or clear everything.

Example

Razor

<div class="datagrid-toolbar">
    <span><strong>@selectedInvoices.Count</strong> selected · total <strong>@FormatMoney(SelectedTotal)</strong></span>
    <button type="button" class="datagrid-button" disabled="@(selectedInvoices.Count == 0)"
            @onclick="ClearInvoices">Clear selection</button>
</div>

<div class="datagrid-frame">
    <BhGrid As="table" aria-label="Invoices" class="datagrid datagrid-selectable"
            SelectionMode="BhGridSelectionMode.Multiple"
            @bind-SelectedKeys="selectedInvoices">
        <BhGridRowGroup As="thead">
            <BhGridRow As="tr" IsHeaderRow="true">
                <BhGridCell As="th" Header="true" class="datagrid-check-cell" OnActivate="ToggleAllInvoices">
                    <BhVisuallyHidden>Select all</BhVisuallyHidden>
                    <span class="datagrid-check" data-checked="@AllInvoicesState" aria-hidden="true"></span>
                </BhGridCell>
                <BhGridCell As="th" Header="true">Invoice</BhGridCell>
                <BhGridCell As="th" Header="true">Customer</BhGridCell>
                <BhGridCell As="th" Header="true">Status</BhGridCell>
                <BhGridCell As="th" Header="true" class="num">Amount</BhGridCell>
            </BhGridRow>
        </BhGridRowGroup>
        <BhGridRowGroup As="tbody">
            @foreach (var invoice in invoices)
            {
                <BhGridRow As="tr" @key="invoice.Id" Key="@invoice.Id">
                    <BhGridCell As="td" class="datagrid-check-cell">
                        <span class="datagrid-check" aria-hidden="true"></span>
                    </BhGridCell>
                    <BhGridCell As="td">@invoice.Id</BhGridCell>
                    <BhGridCell As="td">@invoice.Customer</BhGridCell>
                    <BhGridCell As="td">
                        <span class="datagrid-badge" data-status="@invoice.Status.ToLowerInvariant()">@invoice.Status</span>
                    </BhGridCell>
                    <BhGridCell As="td" class="num">@FormatMoney(invoice.Amount)</BhGridCell>
                </BhGridRow>
            }
        </BhGridRowGroup>
    </BhGrid>
</div>

@code {
    private sealed record Invoice(string Id, string Customer, string Status, decimal Amount);

    private readonly Invoice[] invoices =
    [
        new("INV-1042", "Acme Corp", "Paid", 1_250.00m),
        new("INV-1043", "Globex", "Pending", 830.50m),
        new("INV-1044", "Initech", "Overdue", 4_200.00m),
        new("INV-1045", "Umbrella", "Paid", 615.75m),
        new("INV-1046", "Hooli", "Pending", 2_980.00m),
        new("INV-1047", "Stark Industries", "Paid", 12_400.00m),
    ];

    private IReadOnlyCollection<string> selectedInvoices = [];

    private decimal SelectedTotal => invoices.Where(i => selectedInvoices.Contains(i.Id)).Sum(i => i.Amount);

    private string AllInvoicesState =>
        selectedInvoices.Count == 0 ? "false" : selectedInvoices.Count == invoices.Length ? "true" : "mixed";

    private void ToggleAllInvoices() =>
        selectedInvoices = selectedInvoices.Count == invoices.Length ? [] : invoices.Select(i => i.Id).ToArray();

    private void ClearInvoices() => selectedInvoices = [];

    private static string FormatMoney(decimal amount) =>
        amount.ToString("C", System.Globalization.CultureInfo.GetCultureInfo("en-US"));
}

TreeGrid

TreeGrid="true" renders role="treegrid". Give each row its Level, HasChildren and ParentKey: rows under a collapsed folder get the hidden attribute and are skipped by the keyboard. On a row's first cell, Right expands it (then moves into the row) and Left collapses it (then jumps to the parent row). The chevron is a BhGridRowToggle: a mouse shortcut, hidden from assistive tech, that toggles the row (synced through @@bind-ExpandedKeys); indentation comes from the --bh-grid-level variable on each row.

Example

        new("project/assets/hero.png", "hero.png", "PNG image", 482_113, 3, "project/assets", false),
        new("project/package.json", "package.json", "JSON", 1_240, 2, "project", false),
        new("project/README.md", "README.md", "Markdown", 4_502, 2, "project", false),
        new("notes.txt", "notes.txt", "Text", 1_024, 1, null, false),
    ];

    private IReadOnlyCollection<string> selectedEntry = [];
    private IReadOnlyCollection<string> expandedFolders = ["project", "project/src"];


    private long SizeOf(FsEntry entry) =>
        entry.IsFolder ? fileSystem.Where(e => e.Parent == entry.Path).Sum(e => SizeOf(e)) : entry.Bytes;

    private static string FormatSize(long bytes) => bytes switch
    {
        < 1024 => $"{bytes} B",
        < 1024 * 1024 => $"{bytes / 1024.0:0.#} KB",
        _ => $"{bytes / (1024.0 * 1024):0.#} MB",
    };
}

Div-based grid (calendar)

BhGrid, BhGridRowGroup, BhGridRow and BhGridCell render <div>s with role="grid" / "rowgroup" / "row" / "gridcell" by default, so any layout works — here each row is a 7-column CSS grid. Arrow keys move by day and week, Home / End go to the start / end of the week, and Enter or a click picks a date through each cell's OnActivate. aria-selected, aria-current and aria-disabled on the cells are plain attributes set by the page.

Example

Razor

<div class="calendar-card">
    <div id="grid-calendar-title" class="calendar-title">@MonthStart.ToString("MMMM yyyy", System.Globalization.CultureInfo.InvariantCulture)</div>
    <BhGrid aria-labelledby="grid-calendar-title" class="calendar">
        <BhGridRowGroup>
            <BhGridRow IsHeaderRow="true" class="calendar-row">
                @foreach (var (shortName, longName) in weekdays)
                {
                    <BhGridCell Header="true" class="calendar-weekday">
                        <abbr title="@longName">@shortName</abbr>
                    </BhGridCell>
                }
            </BhGridRow>
        </BhGridRowGroup>
        <BhGridRowGroup class="calendar-weeks">
            @foreach (var week in Weeks)
            {
                <BhGridRow class="calendar-row">
                    @foreach (var day in week)
                    {
                        <BhGridCell class="calendar-day"
                                    aria-selected="@(day == pickedDate ? "true" : null)"
                                    aria-current="@(day == today ? "date" : null)"
                                    aria-disabled="@(day.Month != MonthStart.Month ? "true" : null)"
                                    OnActivate="() => PickDate(day)">@day.Day</BhGridCell>
                    }
                </BhGridRow>
            }
        </BhGridRowGroup>
    </BhGrid>
</div>

<p class="datagrid-output">
    Picked: <strong>@(pickedDate?.ToString("dddd, MMMM d", System.Globalization.CultureInfo.InvariantCulture) ?? "(none)")</strong>
</p>

@code {
    private readonly DateOnly today = DateOnly.FromDateTime(DateTime.Today);
    private DateOnly? pickedDate;

    private readonly (string Short, string Long)[] weekdays =
    [
        ("Su", "Sunday"), ("Mo", "Monday"), ("Tu", "Tuesday"), ("We", "Wednesday"),
        ("Th", "Thursday"), ("Fr", "Friday"), ("Sa", "Saturday"),
    ];

    private DateOnly MonthStart => new(today.Year, today.Month, 1);

    // Whole weeks (Sunday → Saturday) covering the current month.
    private IEnumerable<DateOnly[]> Weeks
    {
        get
        {
            var monthEnd = MonthStart.AddMonths(1).AddDays(-1);
            for (var start = MonthStart.AddDays(-(int)MonthStart.DayOfWeek); start <= monthEnd; start = start.AddDays(7))
            {
                var weekStart = start;
                yield return Enumerable.Range(0, 7).Select(i => weekStart.AddDays(i)).ToArray();
            }
        }
    }

    // Days of the neighbouring months are aria-disabled and can't be picked.
    private void PickDate(DateOnly day)
    {
        if (day.Month == MonthStart.Month) pickedDate = day;
    }
}
